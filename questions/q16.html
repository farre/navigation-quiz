function run() {
  const log = document.querySelector('#out');
  navigation.onnavigate = e => {
    e.intercept({
      async handler() {
        console.log(`intercepted ${e.destination.url}`);
      }
    })
  };
  navigation.onnavigatesuccess = () => {
    console.log("success");
  };
  navigation.onnavigateerrir = () => {
    console.log("error");
  };

  const {committed, finished} = navigation.navigate("document");

  committed.then(() => {
    console.log("committed");
  });

  finished.then(() => {
    console.log("finished");
  });
}
