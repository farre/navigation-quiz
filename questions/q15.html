function log(str) {
  out.innerText += `${str}\n`
}
function run() {
  navigation.onnavigate = e => {
    e.intercept({
      async handler() {
        log(`intercepted ${e.destination.url}`);
        await new Promise(resolve => setTimeout(resolve, 25));
      }
    })
  };
  navigation.onnavigatesuccess = () => {
    log("success");
  };
  navigation.onnavigateerrir = () => {
    log("error");
  };

  const {committed, finished} = navigation.navigate("document");

  committed.then(() => {
    log("committed");
  });

  finished.then(() => {
    log("finished");
  });
}
